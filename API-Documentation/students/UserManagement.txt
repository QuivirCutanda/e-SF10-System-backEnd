# Prerequisites for all endpoints
  - Authorization
    - Headers:
      Key: Authorization
      Value: Bearer your_token_here
  - All endpoints require 'manage_users' permission

--------------------------------------------------
## 1. View All Users
--------------------------------------------------
- Method: GET
- Endpoint: http://localhost:3001/esf10/users

Response (200 OK):
[
  {
    "user_id": 1,
    "first_name": "vincent",
    "middle_name": "admin",
    "last_name": "autentico",
    "email": "admin@example.com",
    "created_at": "2023-05-01T10:00:00.000Z",
    "roles": "admin"
  },
  {
    "user_id": 2,
    "first_name": "Vincent",
    "middle_name": null,
    "last_name": "Autentico",
    "email": "john@example.com",
    "created_at": "2023-05-02T11:30:00.000Z",
    "roles": "teacher"
  }
]

--------------------------------------------------
## 2. Get User by ID
--------------------------------------------------
- Method: GET
- Endpoint: http://localhost:3001/esf10/users/1

Response (200 OK):
{
  "user_id": 1,
  "first_name": "admin",
  "middle_name": "admin",
  "last_name": "admin",
  "email": "admin@example.com",
  "created_at": "2023-05-01T10:00:00.000Z",
  "roles": "admin"
}

Response (404 Not Found):
{
  "message": "User not found"
}

--------------------------------------------------
## 3. Update User
--------------------------------------------------
- Method: PUT
- Endpoint: http://localhost:3001/esf10/users/1
- Body (raw JSON):
{
  "first_name": "vincent",
  "middle_name": "Super",
  "last_name": "Autentico",
  "email": "admin@example.com"
}

Response (200 OK):
{
  "message": "User updated successfully"
}

Response (404 Not Found):
{
  "message": "User not found"
}

--------------------------------------------------
## 4. Delete User
--------------------------------------------------
- Method: DELETE
- Endpoint: http://localhost:3001/esf10/users/2

Response (200 OK):
{
  "message": "User deleted successfully"
}

Response (400 Bad Request):
{
  "message": "Cannot delete your own account"
}

Response (404 Not Found):
{
  "message": "User not found"
}

--------------------------------------------------
## 5. Change User Role
--------------------------------------------------
- Method: PUT
- Endpoint: http://localhost:3001/esf10/users/2/roles
- Body (raw JSON):
{
  "roles": ["teacher", "registrar"]
}

Response (200 OK):
{
  "message": "User roles updated successfully"
}

Response (400 Bad Request):
{
  "message": "Roles must be a non-empty array"
}
or
{
  "errors": [
    {
      "type": "field",
      "value": ["invalid_role"],
      "msg": "Contains invalid role(s). Valid roles are: admin, teacher, student, registrar",
      "path": "roles",
      "location": "body"
    }
  ]
}

Response (404 Not Found):
{
  "message": "User not found"
}

--------------------------------------------------
## 6. Search Users
--------------------------------------------------
- Method: GET
- Endpoint: http://localhost:3001/esf10/users/search?query=your_search_term
- Query Parameters:
  query: Search term to find users by name or email

Response (200 OK):
